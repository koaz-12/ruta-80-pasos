🔍 INSPECTOR DE CASILLAS

Instrucciones:

  1. Abre el juego en otra pestaña
  2. Presiona F12 para abrir la consola
  3. Copia el código de abajo
  4. Pégalo en la consola y presiona Enter
  5. Haz click en cualquier casilla del tablero

Código para pegar en consola:

// INSPECTOR DIRECTO - v7.7
(function() {
    console.clear();
    console.log('%c🔍 INSPECTOR ACTIVADO', 'color:cyan; font-size:20px; font-weight:bold');
    
    document.addEventListener('click', function(e) {
        let el = e.target;
        
        // Buscar grupo de casilla
        for (let i = 0; i < 10; i++) {
            if (!el) break;
            
            const dataId = el.getAttribute('data-id');
            if (dataId) {
                // Crear overlay
                let ov = document.getElementById('quick-inspector');
                if (!ov) {
                    ov = document.createElement('div');
                    ov.id = 'quick-inspector';
                    document.body.appendChild(ov);
                }
                
                const x = el.dataset.x || '?';
                const y = el.dataset.y || '?';
                const disp = el.getAttribute('data-display') || dataId;
                
                ov.innerHTML = `
                    <div style="position:fixed; top:50%; left:50%; transform:translate(-50%,-50%);
                               background:black; color:white; padding:40px; border:4px solid cyan;
                               border-radius:15px; font-family:monospace; font-size:18px; z-index:999999;
                               min-width:350px; box-shadow:0 0 50px rgba(0,255,255,0.5);">
                        <h1 style="text-align:center; color:cyan; margin:0 0 20px 0;">CASILLA ${dataId}</h1>
                        <div style="line-height:2;">
                            <div><b style="color:yellow;">ID:</b> ${dataId}</div>
                            <div><b style="color:yellow;">Display:</b> ${disp}</div>
                            <div><b style="color:yellow;">X:</b> ${x}</div>
                            <div><b style="color:yellow;">Y:</b> ${y}</div>
                        </div>
                        <div style="text-align:center; margin-top:20px;">
                            <button onclick="this.parentElement.parentElement.parentElement.remove()"
                                    style="background:cyan; color:black; border:none; padding:15px 40px;
                                           font-size:18px; font-weight:bold; cursor:pointer; border-radius:8px;">
                                CERRAR
                            </button>
                        </div>
                    </div>
                `;
                
                console.log('%c━━━━ TILE DATA ━━━━', 'color:yellow');
                console.log('ID:', dataId, '| X:', x, '| Y:', y);
                return;
            }
            el = el.parentElement;
        }
    });
    
    alert('✅ Inspector activado!\\n\\nHaz click en cualquier casilla del tablero.');
})();
